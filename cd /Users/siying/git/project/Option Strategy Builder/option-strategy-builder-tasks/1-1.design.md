### 3. 游戏状态管理架构

```typescript
// src/stores/gameStore.ts
import { create } from 'zustand'
import { devtools } from 'zustand/middleware'

interface GameStore {
  // 游戏状态
  gameState: GameState
  isRolling: boolean
  isMoving: boolean
  showMarketEvent: boolean
  
  // 游戏操作
  initGame: () => void
  rollDice: () => Promise<number>
  movePlayer: (steps: number) => Promise<void>
  selectStrategy: (strategyId: string) => Promise<StrategyResult>
  endTurn: () => void
  
  // 状态更新
  updatePlayerFunds: (amount: number) => void
  updatePlayerPosition: (position: number) => void
  triggerMarketEvent: (scenario: MarketScenario) => void
  completeMarketEvent: () => void
}

export const useGameStore = create<GameStore>()(
  devtools(
    (set, get) => ({
      // 初始状态
      gameState: {
        gameId: '',
        status: 'waiting',
        currentTurn: 0,
        maxTurns: 20,
        player: {
          id: 'player-1',
          name: '新手交易员',
          avatar: 'default',
          position: 0,
          funds: 10000,
          completedStrategies: [],
          achievements: []
        },
        board: BOARD_SPACES
      },
      isRolling: false,
      isMoving: false,
      showMarketEvent: false,

      // 游戏操作实现
      initGame: () => {
        set((state) => ({
          gameState: {
            ...state.gameState,
            gameId: `game-${Date.now()}`,
            status: 'playing',
            currentTurn: 0
          }
        }))
      },

      rollDice: async () => {
        set({ isRolling: true })
        
        // 模拟掷骰子动画
        await new Promise(resolve => setTimeout(resolve, 1000))
        
        const dice1 = Math.floor(Math.random() * 6) + 1
        const dice2 = Math.floor(Math.random() * 6) + 1
        const total = dice1 + dice2
        
        set({ isRolling: false })
        return total
      },

      movePlayer: async (steps: number) => {
        const { gameState } = get()
        const newPosition = (gameState.player.position + steps) % 12
        
        set({ isMoving: true })
        
        // 移动动画
        await new Promise(resolve => setTimeout(resolve, 1500))
        
        set((state) => ({
          gameState: {
            ...state.gameState,
            player: {
              ...state.gameState.player,
              position: newPosition
            }
          },
          isMoving: false
        }))
        
        // 触发格子效果
        const space = BOARD_SPACES[newPosition]
        if (space.marketScenario) {
          get().triggerMarketEvent(space.marketScenario)
        }
      },

      selectStrategy: async (strategyId: string) => {
        const { gameState } = get()
        const currentSpace = BOARD_SPACES[gameState.player.position]
        
        // 调用策略计算服务
        const result = await calculateStrategyResult(
          strategyId,
          currentSpace.marketScenario!
        )
        
        // 更新玩家资金
        get().updatePlayerFunds(result.profit)
        
        return result
      },

      // 其他方法实现...
    }),
    { name: 'game-store' }
  )
)
```

export const BOARD_SPACES: BoardSpace[] = [

    icon: '📈',
    marketScenario: {
      id: 'bull-1',
      name: '经典牛市',
      description: 'BTC从$45,000涨至$50,000',
      priceChange: 0.11,
      volatilityChange: -0.05,
      timeToExpiry: 7,
      recommendedStrategy: ['long-call', 'bull-spread'],
      difficulty: 'easy'
    }
  },
  // ... 其他10个格子的配置
]

export class GameLogic {
  static calculateTurnResult(
    strategy: OptionStrategy,
    marketScenario: MarketScenario,
    currentPrice: number
  ): StrategyResult {
    // 期权损益计算逻辑
    const spotAtExpiry = currentPrice * (1 + marketScenario.priceChange)
    let totalPnL = 0
    
    for (const position of strategy.positions) {
      const positionPnL = this.calculatePositionPnL(
        position,
        spotAtExpiry
      )
      totalPnL += positionPnL
    }
    
    // 判断是否为最优策略
    const isOptimal = marketScenario.recommendedStrategy.includes(strategy.id)
    
    return {
      profit: totalPnL,
      explanation: this.generateExplanation(strategy, marketScenario, isOptimal),
      isOptimal,
      alternativeStrategies: this.getAlternativeAnalysis(marketScenario)
    }
  }
  
  private static calculatePositionPnL(
    position: OptionPosition,
    spotAtExpiry: number
  ): number {
    let intrinsicValue = 0
    
    if (position.type === 'call') {
      intrinsicValue = Math.max(0, spotAtExpiry - position.strike)
    } else {
      intrinsicValue = Math.max(0, position.strike - spotAtExpiry)
    }
    
    if (position.action === 'long') {
      return position.quantity * (intrinsicValue - position.premium)
    } else {
      return position.quantity * (position.premium - intrinsicValue)
    }
  }
}
